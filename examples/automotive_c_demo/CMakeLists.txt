cmake_minimum_required(VERSION 3.16)
project(automotive_c_demo C)
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_C_EXTENSIONS OFF)
add_library(automotive_c_core src/sensor_service.c src/app.c)
target_include_directories(automotive_c_core PUBLIC include)
if(CMAKE_C_COMPILER_ID MATCHES "GNU|Clang")
  target_compile_options(automotive_c_core PRIVATE -Wall -Wextra -Wpedantic -Wconversion -Wshadow)
endif()
add_library(automotive_c_host_fake platform/host/fake_sensor.c)
target_include_directories(automotive_c_host_fake PUBLIC include platform/host)
target_link_libraries(automotive_c_host_fake PUBLIC automotive_c_core)
add_executable(test_automotive_c_demo tests/test_main.c)
target_include_directories(test_automotive_c_demo PRIVATE platform/host)
target_link_libraries(test_automotive_c_demo PRIVATE automotive_c_host_fake automotive_c_core)
enable_testing()
add_test(NAME automotive_c_demo_tests COMMAND test_automotive_c_demo)
